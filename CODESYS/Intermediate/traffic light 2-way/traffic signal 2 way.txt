elapsed_time:= traffic_timer.ET;
NorthSouthGreen:= start AND NOT NorthSouthRed AND NOT NorthSouthYellow;
EastWestRed:= start AND NOT EastWestGreen AND NOT EastWestYellow;
traffic_timer (in:= (start AND NOT stop AND NOT traffic_timer.Q), pt:= timer_preset);
    // North-South Green Phase
IF NorthSouthGreen AND elapsed_time >= green_preset AND elapsed_time < timer_preset THEN
	NorthSouthYellow:= TRUE;
	NorthSouthGreen:= FALSE;
END_IF
 // North-South Yellow Phase
IF NorthSouthYellow AND elapsed_time >= timer_preset THEN
	NorthSouthYellow:= FALSE;
	NorthSouthRed:= TRUE;
	EastWestGreen:= TRUE;
END_IF
// East-West Green Phase
IF EastWestGreen AND elapsed_time >= green_preset AND elapsed_time < timer_preset THEN
	EastWestYellow:= TRUE;
	EastWestGreen:= FALSE;
END_IF
 // East-West Yellow Phase
IF EastWestYellow AND elapsed_time >= timer_preset THEN
	EastWestYellow:= FALSE;
	NorthSouthRed:= FALSE;
END_IF
