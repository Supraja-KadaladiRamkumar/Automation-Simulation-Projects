//zone 1 Storage unit

IF bStart AND iStage = 0 AND NOT bStop THEN
	bCycleStart := ((bStart OR bCycleStart) AND NOT bStop);
	iStage := 1;
END_IF

// use separate alarm acknowledgement variable for manually resetting when obstruction happens

IF (bCycleStart AND iStage = 1) AND NOT bStop AND NOT bZone2Ready AND NOT bZone3Running THEN
	bZone1Running := TRUE;

	IF (bZone1Running AND (rZone1Temp > rZone1TempMax OR rZone1Temp < rZone1TempMin) OR bDoorOpenZone1 OR bIntrusionZone1) THEN
		bZone1Ready := FALSE;
		bAlarmZone1 := TRUE;
	ELSE
		bAlarmZone1 := FALSE;
		bZone1Ready := TRUE;
		bPackingStarted := FALSE;
	END_IF
ELSE
	bAlarmZone1 := FALSE;
	bZone1Ready := FALSE;
	rZone1Temp := 0.0;
END_IF

 //zone 2 processing unit
IF iStage = 2 AND bMachineMalfunction THEN
	bFaultZone2 := TRUE;      // latch fault
	bMachineRunning := FALSE;
END_IF;
IF iStage = 2 AND bHumanReset THEN
	bFaultZone2 := FALSE;     // clear fault on human reset
	bMachineRunning := TRUE;
END_IF;

IF (bZone1Ready OR (bCycleStart AND iStage = 2 AND bMachineRunning)) AND NOT bStop AND NOT bAlarmZone1 AND NOT bZone3Ready AND NOT bFaultZone2 THEN
	tmrTransferZone12(IN := (iStage = 1) AND bZone1Running AND NOT bZone2Running, PT := T#10S);
	bMachineRunning := TRUE;
	IF tmrTransferZone12.Q THEN
		bZone2Running := TRUE;
		iStage := 2;
		bZone1Ready := FALSE;
		bZone1Running := FALSE;
	END_IF
	IF bMachineRunning THEN
		tmrProcess(IN := bZone2Running AND NOT bFaultZone2 AND NOT bAlarmZone1 AND NOT bHumanReset,PT := T#15S);
		tProcessET := tmrProcess.ET;
		IF tmrProcess.Q THEN
			bBatchComplete := TRUE;
			bZone2Ready := TRUE;
		ELSE
			bBatchComplete := FALSE;
			bZone2Ready := FALSE;
		END_IF
	ELSE
		bBatchComplete := FALSE;
		bZone2Ready := FALSE;
	END_IF
END_IF
 
//zone 3 packaging unit
IF (bZone2Ready OR (bCycleStart AND iStage = 3 AND bPackingStarted)) AND NOT bStop AND NOT bZone1Ready AND NOT bFaultZone2 THEN
	tmrTransferZone23( IN := (iStage = 2) AND bZone2Running AND NOT bZone3Running, PT := T#10S);
	IF tmrTransferZone23.Q THEN
		bZone3Running := TRUE;
		iStage := 3;
		bZone2Ready := FALSE;
		bZone2Running := FALSE;
		bBatchComplete := FALSE;
		bMachineRunning := FALSE;
		bPackingStarted := TRUE;
	END_IF
	cntPackaging(CU := bObjectDetectSensor AND bZone3Running, PV := 3, RESET := bCounterReset);
	iCounterCV := cntPackaging.CV;
	IF cntPackaging.CV = cntPackaging.PV THEN
		bZone3Ready := TRUE;
	ELSIF cntPackaging.CV > cntPackaging.PV THEN
		bZone3Ready := FALSE;
	END_IF
	tmrTransferZone31(IN := (iStage = 3) AND bZone3Ready AND NOT bZone1Running, PT := T#10S);
	IF tmrTransferZone31.Q THEN
		bCounterReset := tmrTransferZone31.Q AND iStage = 3;
		bZone3Running := FALSE;
		bZone3Ready := FALSE;
		iStage := 1;
		bPackingStarted := FALSE;
	ELSE
		bCounterReset := FALSE;
	END_IF
END_IF
